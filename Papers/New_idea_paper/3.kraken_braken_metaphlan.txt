  set -uo pipefail

  INPUT_DIR="/mnt/data/Metagenomics/omics_logic/host_decontamination/kneaddata"
  OUT_DIR="/mnt/data/Metagenomics/omics_logic/kraken2_results"
  THREADS=64

  REPORT_DIR="$OUT_DIR/reports"
  LOG_DIR="$OUT_DIR/logs"
  STATUS_DIR="$OUT_DIR/status"

  mkdir -p "$REPORT_DIR" "$LOG_DIR" "$STATUS_DIR"

  for tool in kraken2; do
      command -v "$tool" >/dev/null 2>&1 || {
          echo "ERROR: $tool is not installed."
          exit 1
      }
  done

  for file in hash.k2d opts.k2d taxo.k2d; do
      [[ -s "$DB/$file" ]] || {
          echo "ERROR: missing database file: $DB/$file"
          exit 1
      }
  done

  shopt -u nullglob

  if ((TOTAL == 0)); then
      echo "ERROR: no KneadData paired FASTQ files found."
      exit 1
  fi

  SKIPPED=0
  FAILED=0

  echo "Found $TOTAL samples."
  echo "Kraken2 database: $DB"
  echo "Threads: $THREADS"

  for i in "${!R1_FILES[@]}"; do
      R1="${R1_FILES[$i]}"
      SAMPLE="$(basename "$R1" _kneaddata_paired_1.fastq)"
      R2="$(dirname "$R1")/${SAMPLE}_kneaddata_paired_2.fastq"

      REPORT="$REPORT_DIR/${SAMPLE}.kraken2.report"
      REPORT_TMP="$REPORT.part"
      LOG="$LOG_DIR/${SAMPLE}.kraken2.log"
      DONE="$STATUS_DIR/${SAMPLE}.done"

      echo
      echo "[$((i + 1))/$TOTAL] $SAMPLE"

      if [[ -f "$DONE" && -s "$REPORT" ]]; then
          echo "[SKIP] Already completed."
          SKIPPED=$((SKIPPED + 1))
          continue
      fi

      if [[ ! -s "$R2" ]]; then
          echo "[FAIL] Missing mate: $R2"
          FAILED=$((FAILED + 1))
          continue
      fi

      rm -f -- "$REPORT_TMP"

      {
          echo "Started: $(date --iso-8601=seconds)"
          echo "R1: $R1"
          echo "R2: $R2"

          kraken2 \
              --db "$DB" \
              --memory-mapping \
              --threads "$THREADS" \
              --paired \
              --use-names \
              --report "$REPORT_TMP" \

  echo "Completed: $COMPLETED"

  echo "Failed:    $FAILED"

#!/usr/bin/env bash

set -uo pipefail

# =============================================================================
# Configuration
# =============================================================================

SCRIPT_DIR="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)"

DB_DIR="$SCRIPT_DIR/../kraken2db"
KRAKEN_ROOT="$SCRIPT_DIR/kraken2_results"
BRACKEN_ROOT="$SCRIPT_DIR/bracken_results"

GROUPS=("CD" "UC" "non_IBD")

# Must correspond to a Bracken file such as:
# database150mers.kmer_distrib
READ_LENGTH=150

# S = species, G = genus, F = family
TAXONOMIC_LEVEL="S"

# Minimum number of Kraken-assigned reads required
THRESHOLD=10

RUN_ID="$(date '+%Y%m%d_%H%M%S')"

mkdir -p "$BRACKEN_ROOT"

MASTER_LOG="$BRACKEN_ROOT/bracken_run_${RUN_ID}.log"

# Log terminal output and save it to the master log
exec > >(tee -a "$MASTER_LOG") 2>&1

echo "============================================================"
echo "Bracken abundance-estimation pipeline"
echo "Started:           $(date --iso-8601=seconds)"
echo "Kraken root:       $KRAKEN_ROOT"
echo "Bracken output:    $BRACKEN_ROOT"
echo "Database:          $DB_DIR"
echo "Read length:       $READ_LENGTH"
echo "Taxonomic level:   $TAXONOMIC_LEVEL"
echo "Read threshold:    $THRESHOLD"
echo "Groups:            ${GROUPS[*]}"
echo "Master log:        $MASTER_LOG"
echo "============================================================"

# =============================================================================
# Preflight checks
# =============================================================================

command -v bracken >/dev/null 2>&1 || {
    echo "ERROR: bracken is not installed or is not available in PATH."
    exit 1
}

for FILE in hash.k2d opts.k2d taxo.k2d; do
    [[ -s "$DB_DIR/$FILE" ]] || {
        echo "ERROR: missing Kraken2 database file: $DB_DIR/$FILE"
        exit 1
    }
done

DISTRIBUTION_FILE="$DB_DIR/database${READ_LENGTH}mers.kmer_distrib"

[[ -s "$DISTRIBUTION_FILE" ]] || {
    echo "ERROR: missing Bracken distribution file:"
    echo "       $DISTRIBUTION_FILE"
    echo
    echo "Available distributions:"
    find "$DB_DIR" -maxdepth 1 \
        -name 'database*mers.kmer_distrib' \
        -printf '  %f\n' | sort
    exit 1
}

# =============================================================================
# Count input reports
# =============================================================================

TOTAL=0

for GROUP in "${GROUPS[@]}"; do
    REPORT_DIR="$KRAKEN_ROOT/$GROUP/reports"

    [[ -d "$REPORT_DIR" ]] || {
        echo "WARNING: report directory not found: $REPORT_DIR"
        continue
    }

    shopt -s nullglob
    REPORTS=("$REPORT_DIR"/*.kraken2.report)
    shopt -u nullglob

    TOTAL=$((TOTAL + ${#REPORTS[@]}))
done

if (( TOTAL == 0 )); then
    echo "ERROR: no Kraken2 report files were found."
    exit 1
fi

echo
echo "Found $TOTAL Kraken2 reports."
echo

# =============================================================================
# Run Bracken
# =============================================================================

PROCESSED=0
COMPLETED=0
SKIPPED=0
FAILED=0
FAILED_SAMPLES=()

for GROUP in "${GROUPS[@]}"; do
    KRAKEN_REPORT_DIR="$KRAKEN_ROOT/$GROUP/reports"

    [[ -d "$KRAKEN_REPORT_DIR" ]] || continue

    OUTPUT_DIR="$BRACKEN_ROOT/$GROUP/species_abundance"
    REPORT_DIR="$BRACKEN_ROOT/$GROUP/reports"
    LOG_DIR="$BRACKEN_ROOT/$GROUP/logs"
    STATUS_DIR="$BRACKEN_ROOT/$GROUP/status"

    mkdir -p \
        "$OUTPUT_DIR" \
        "$REPORT_DIR" \
        "$LOG_DIR" \
        "$STATUS_DIR"

    shopt -s nullglob
    KRAKEN_REPORTS=("$KRAKEN_REPORT_DIR"/*.kraken2.report)
    shopt -u nullglob

    if (( ${#KRAKEN_REPORTS[@]} == 0 )); then
        echo "WARNING: no Kraken2 reports found for group: $GROUP"
        continue
    fi

    echo "------------------------------------------------------------"
    echo "Processing group: $GROUP"
    echo "Reports found:    ${#KRAKEN_REPORTS[@]}"
    echo "------------------------------------------------------------"

    for KRAKEN_REPORT in "${KRAKEN_REPORTS[@]}"; do
        PROCESSED=$((PROCESSED + 1))

        SAMPLE="$(basename "$KRAKEN_REPORT" .kraken2.report)"

        BRACKEN_OUTPUT="$OUTPUT_DIR/${SAMPLE}.bracken.${TAXONOMIC_LEVEL}.tsv"
        BRACKEN_REPORT="$REPORT_DIR/${SAMPLE}.bracken.${TAXONOMIC_LEVEL}.report"

        OUTPUT_TMP="${BRACKEN_OUTPUT}.part"
        REPORT_TMP="${BRACKEN_REPORT}.part"

        LOG="$LOG_DIR/${SAMPLE}.bracken.log"
        DONE="$STATUS_DIR/${SAMPLE}.done"

        echo
        echo "[$PROCESSED/$TOTAL] $GROUP/$SAMPLE"

        if [[ -f "$DONE" &&
              -s "$BRACKEN_OUTPUT" &&
              -s "$BRACKEN_REPORT" ]]; then
            echo "[SKIP] Already completed."
            SKIPPED=$((SKIPPED + 1))
            continue
        fi

        if [[ ! -s "$KRAKEN_REPORT" ]]; then
            echo "[FAIL] Empty Kraken2 report: $KRAKEN_REPORT"
            FAILED=$((FAILED + 1))
            FAILED_SAMPLES+=("$GROUP/$SAMPLE")
            continue
        fi

        rm -f -- "$OUTPUT_TMP" "$REPORT_TMP"

        {
            echo "Sample:            $SAMPLE"
            echo "Group:             $GROUP"
            echo "Started:           $(date --iso-8601=seconds)"
            echo "Kraken report:     $KRAKEN_REPORT"
            echo "Bracken database:  $DB_DIR"
            echo "Read length:       $READ_LENGTH"
            echo "Taxonomic level:   $TAXONOMIC_LEVEL"
            echo "Threshold:         $THRESHOLD"
            echo

            bracken \
                -d "$DB_DIR" \
                -i "$KRAKEN_REPORT" \
                -o "$OUTPUT_TMP" \
                -w "$REPORT_TMP" \
                -r "$READ_LENGTH" \
                -l "$TAXONOMIC_LEVEL" \
                -t "$THRESHOLD"

            STATUS=$?

            echo
            echo "Bracken exit status: $STATUS"
            echo "Finished: $(date --iso-8601=seconds)"

            exit "$STATUS"
        } >"$LOG" 2>&1

        STATUS=$?

        if [[ $STATUS -eq 0 &&
              -s "$OUTPUT_TMP" &&
              -s "$REPORT_TMP" ]]; then

            mv -- "$OUTPUT_TMP" "$BRACKEN_OUTPUT"
            mv -- "$REPORT_TMP" "$BRACKEN_REPORT"
            : > "$DONE"

            echo "[OK] Abundance table: $BRACKEN_OUTPUT"
            echo "[OK] Bracken report:  $BRACKEN_REPORT"

            COMPLETED=$((COMPLETED + 1))
        else
            echo "[FAIL] Bracken failed. Check: $LOG"

            FAILED=$((FAILED + 1))
            FAILED_SAMPLES+=("$GROUP/$SAMPLE")
        fi
    done
done

# =============================================================================
# Final summary
# =============================================================================

echo
echo "============================================================"
echo "Bracken pipeline summary"
echo "Finished:  $(date --iso-8601=seconds)"
echo "Total:     $TOTAL"
echo "Completed: $COMPLETED"
echo "Skipped:   $SKIPPED"
echo "Failed:    $FAILED"
echo "Master log: $MASTER_LOG"
echo "============================================================"

if (( FAILED > 0 )); then
    echo
    echo "Failed samples:"

    for SAMPLE in "${FAILED_SAMPLES[@]}"; do
        echo "  $SAMPLE"
    done

    exit 1
fi

echo "Pipeline finished successfully."
exit 0

The most important distinction is:

bracken -t 10         = read threshold
bracken-build -t 64   = build threads

Bracken itself does not have a normal multithreading flag. It operates on the Kraken report and redistributes reads assigned at higher taxonomic levels to the requested level using the database’s expected k-mer distributions. It does not read the FASTQ files or rerun classification.

For your post-fastp reads, -r 150 is appropriate if their average retained length remains close to 150 bp. If the average is closer to 125, 100, or 75 bp, use the nearest corresponding distribution file available in your database.

The easiest method is to use the fastp JSON reports. They contain post-filtering mean lengths for R1 and R2, plus total reads and bases. fastp documents JSON report generation with --json.

First locate the reports:

find host_decontamination -type f -name '*.json' -printf '%p\n'

For one sample:

jq '.summary.after_filtering |
    {
      total_reads,
      total_bases,
      read1_mean_length,
      read2_mean_length,
      combined_mean_length: (.total_bases / .total_reads)
    }' sample.fastp.json

Example interpretation:

read1_mean_length:   147
read2_mean_length:   145
combined_mean_length: 146

In that case, 150 is the closest Bracken length.

For all fastp reports together, calculate a read-count-weighted mean:

FASTP_DIR="/home/bioinfo/Desktop/Data_5TB/Metagenomics/omics_logic/host_decontamination"

mapfile -d '' JSON_FILES < <(
    find "$FASTP_DIR" -type f -name '*.json' -print0
)

if (( ${#JSON_FILES[@]} == 0 )); then
    echo "No fastp JSON reports found."
else
    MEAN_LENGTH=$(
        jq -s '
          ([.[].summary.after_filtering.total_bases] | add) /
          ([.[].summary.after_filtering.total_reads] | add)
        ' "${JSON_FILES[@]}"
    )

    printf 'Weighted post-fastp mean length: %.2f bp\n' "$MEAN_LENGTH"
fi

            echo "[OK] Abundance table: $BRACKEN_OUTPUT"
            echo "[OK] Bracken report:  $BRACKEN_REPORT"
            echo "[OK] Log:             $LOG"

            COMPLETED=$((COMPLETED + 1))
        else
            echo "[FAIL] Bracken failed."
            echo "       Check log: $LOG"

            FAILED=$((FAILED + 1))
            FAILED_SAMPLES+=("$COHORT/$SAMPLE")
        fi
    done
done

# =============================================================================
# Final summary
# =============================================================================

echo
echo "============================================================"
echo "Bracken pipeline summary"
echo "Finished:   $(date --iso-8601=seconds)"
echo "Total:      $TOTAL"
echo "Processed:  $PROCESSED"
echo "Completed:  $COMPLETED"
echo "Skipped:    $SKIPPED"
echo "Failed:     $FAILED"
echo "Master log: $MASTER_LOG"
echo "============================================================"

if (( FAILED > 0 )); then
    echo
    echo "Failed samples:"

    for FAILED_SAMPLE in "${FAILED_SAMPLES[@]}"; do
        echo "  $FAILED_SAMPLE"
    done

    exit 1
fi

echo "Pipeline finished successfully."
exit 0

 #!/usr/bin/env bash
set -Eeuo pipefail

# Optional first argument overrides the database location.
DB_ROOT="${1:-/home/bioinfo/Desktop/Data_5TB/Metagenomics/HUMAnN3/databases}"
THREADS=16
MPA_INDEX="mpa_vJun23_CHOCOPhlAnSGB_202403"

LOG_DIR="$DB_ROOT/logs"
STATUS_DIR="$DB_ROOT/status"
mkdir -p "$LOG_DIR" "$STATUS_DIR" "$DB_ROOT/metaphlan"

LOG="$LOG_DIR/download_$(date '+%Y%m%d_%H%M%S').log"
exec > >(tee -a "$LOG") 2>&1

trap 'echo "[ERROR] Failed at line $LINENO. See: $LOG"' ERR
trap 'echo "[STOPPED] Rerun this script to retry unfinished steps."; exit 130' INT
trap 'echo "[STOPPED] Download terminated."; exit 143' TERM

for TOOL in humann humann_databases humann_config metaphlan bowtie2-build; do
    command -v "$TOOL" >/dev/null 2>&1 || {
        echo "ERROR: $TOOL not found. Activate the human environment."
        exit 1
    }
done

echo "=== HUMAnN3 database installation ==="
echo "Started:   $(date --iso-8601=seconds)"
echo "Directory: $DB_ROOT"
echo "Log:       $LOG"
humann --version
metaphlan --version
df -h "$DB_ROOT"

# Successful steps are recorded and skipped on subsequent runs.
run_step() {
    local STEP="$1"
    shift
    local MARKER="$STATUS_DIR/${STEP}.done"

    if [[ -f "$MARKER" ]]; then
        echo "[SKIP] $STEP previously completed."
        return 0
    fi

    echo
    echo "[START] $STEP: $(date --iso-8601=seconds)"
    "$@"
    touch "$MARKER"
    echo "[OK] $STEP: $(date --iso-8601=seconds)"
}

run_step "metaphlan_${MPA_INDEX}" \
    metaphlan --install \
    --bowtie2db "$DB_ROOT/metaphlan" \
    --index "$MPA_INDEX" \
    --nproc "$THREADS"

run_step "chocophlan_full_v201901_v31" \
    humann_databases --download chocophlan full "$DB_ROOT"

# Full UniRef90, not the smaller EC-filtered subset.
run_step "uniref90_full_v201901b" \
    humann_databases --download uniref uniref90_diamond "$DB_ROOT"

run_step "utility_mapping_full_v201901b" \
    humann_databases --download utility_mapping full "$DB_ROOT"

# Configure paths explicitly, including when completed steps were skipped.
humann_config --update database_folders nucleotide "$DB_ROOT/chocophlan"
humann_config --update database_folders protein "$DB_ROOT/uniref"
humann_config --update database_folders utility_mapping "$DB_ROOT/utility_mapping"

echo
humann_config --print
echo "=== All database installation steps completed ==="
echo "Finished: $(date --iso-8601=seconds)"
echo "Log: $LOG"
